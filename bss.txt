def insert_event_store(event_store: EventStore):
    """
    Inserts an event store record into the database if not exists.
    """
    try:
        with Session() as db:
            existing = db.query(EventStore).filter_by(request_reference=event_store.request_reference).first()
            if existing:
                logger.info(f"EventStore with request_reference={event_store.request_reference} already exists. Skipping insert.")
                return existing
            
            db.add(event_store)
            db.commit()
            db.refresh(event_store)
            logger.info(
                f"Successfully inserted event store record with ID: {event_store.id if hasattr(event_store, 'id') else 'N/A'}"
            )
            return event_store
    except Exception as e:
        logger.error(f"Error inserting event store record: {str(e)}")
        raise
